# scripts/publish_wordpress.py
import os
import requests
from requests.auth import HTTPBasicAuth

# 사용자 > 프로필 > 애플리케이션 비밀번호 에서 발급 (WordPress 5.6 이상)
# 발급한 비밀번호는 코드에 적지 말고 환경 변수로 둡니다
WP_URL = os.environ['WP_URL']            # https://yoursite.com
WP_USER = os.environ['WP_USER']
WP_APP_PASSWORD = os.environ['WP_APP_PASSWORD']

def publish_to_wordpress(title, content, status='draft', categories=None):
    """WordPress REST API 로 글을 올린다.

    status 옵션:
    - 'draft'  : 임시저장 (권장 — 검토 후 직접 발행)
    - 'publish': 즉시 발행
    """
    r = requests.post(
        f'{WP_URL}/wp-json/wp/v2/posts',
        json={'title': title, 'content': content, 'status': status,
              'categories': categories or []},
        auth=HTTPBasicAuth(WP_USER, WP_APP_PASSWORD),
        timeout=30)

    if r.status_code == 201:
        url = r.json()['link']
        print(f' 발행 완료: {url}')
        return url
    print(f' 발행 실패 ({r.status_code}): {r.text[:200]}')
    return None
