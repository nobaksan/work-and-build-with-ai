# ~/.bashrc 또는 ~/.zshrc에 추가
alias brief="python ~/personal-assistant/scripts/daily_briefing.py"
alias todo="python ~/personal-assistant/scripts/todo.py"
alias cal="python ~/personal-assistant/scripts/calendar_sync.py"

# 사용:

cal today      # 오늘 일정
mail           # 이메일 요약
